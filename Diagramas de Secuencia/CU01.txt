@startuml
 
actor Conserje
participant "UI- Login" as UIL
participant "Gestor de Usuario" as GDU
participant "UsuarioDAO" as UsDAO
Conserje -> UIL: Desea logearse
loop mientras no logre logearse
activate UIL
UIL -> Conserje: Muestra Campos 
deactivate UIL
Conserje -> UIL:Ingresa Datos
activate UIL
UIL -> UIL: Verifica CamposL
opt campos obligarorios incompletos o formato de campos erroneo
  UIL -> Conserje: Muestra mensajes sobre campos incompletos
end
UIL -> GDU: autenticarUsuario(usuarioDTO)
deactivate UIL
activate GDU
GDU -> UsDAO: autenticarUsuario(usuarioDTO)
deactivate GDU
activate UsDAO

alt Existe usuario
UsDAO -> GDU: Existe Usuario
deactivate UsDAO
activate GDU
GDU -> UIL: Existe Usuario
deactivate GDU
activate UIL
UIL -> Conserje: Viaja a otro CU (pantalla principal)
deactivate UIL

else No existe usuario
activate UsDAO
UsDAO -> GDU: No existe usuario
deactivate UsDAO
activate GDU
GDU -> UIL: No existe usuario
deactivate GDU
activate UIL
UIL -> Conserje: MUestra mensaje de usuario o contraseña incorrecto
deactivate UIL
end

end
@enduml
