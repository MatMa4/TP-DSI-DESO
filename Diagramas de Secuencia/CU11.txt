@startuml

actor Conserje
participant "UI- Dar baja de Huesped" as UIH
participant "Gestor De Huesped" as GDH
participant "HuespedDAO" as HuesDAO
participant "HuespedDTO" as HuesDTO
    Conserje -> UIH: Presiona la tecla "BORRAR"
    activate UIH
    UIH -> GDH: consultarDocumento(NroDocumento)
    deactivate UIH
    activate GDH
    GDH -> HuesDAO: consultarDocumento(NroDocumento)
    deactivate GDH
    activate HuesDAO
    HuesDAO -> HuesDTO: buscarDatos()
    alt huesped jamás alojado
        deactivate HuesDAO
        activate HuesDTO
        HuesDTO -> HuesDAO: HuespedDTO
        deactivate HuesDTO
        activate HuesDAO
        HuesDAO -> GDH: HuespedDTO
        deactivate HuesDAO
        activate GDH
        GDH -> UIH: HuespedDTO
        deactivate GDH
        activate UIH
        UIH -> Conserje: Mensaje: Este huesped será eliminado
        deactivate UIH
    else huesped alojado anteriormente
        activate HuesDTO
        HuesDTO -> HuesDAO: HuespedDTO
        deactivate HuesDTO
        activate HuesDAO
        HuesDAO -> GDH: HuespedDTO
        deactivate HuesDAO
        activate GDH
        GDH -> UIH: HuespedDTO
        deactivate GDH
        activate UIH
        UIH -> Conserje: Mensaje: Este huesped no se puede borrar
        deactivate UIH
    end
    alt conserje desea eliminar
    Conserje -> UIH: presiona "ELIMINAR"
    activate UIH
    UIH -> GDH: HuespedDTO
    deactivate UIH
    activate GDH
    GDH -> Huesped: eliminarHuesped(HuespedDTO)
    activate Huesped
    Huesped -> GDH
    deactivate Huesped
    GDH -> UIH: Datos eliminados correctamente
    deactivate GDH
    activate UIH
    UIH -> Conserje: Datos eliminados del sistema
    deactivate UIH
    else presiona cancelar
    Conserje -> UIH: presiona "CANCELAR"
    activate UIH
    deactivate UIH
    end

@enduml
